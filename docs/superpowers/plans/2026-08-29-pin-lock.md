# PIN Lock with 2-Minute Idle Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Lock the app after 2 minutes of inactivity with a 6-digit PIN the user sets on first login. Coexists with the existing biometric lock.

**Architecture:** `PinService` (PBKDF2 + secure storage), `LockStateProvider` (state machine), `IdleLockService` (timer + lifecycle), and three new screens (`CreatePinScreen`, `PinLockScreen`, `PinLockSettingsScreen`). The settings page relabels the existing 2FA entry to "PIN lock" using the same icon.

**Tech Stack:** Flutter 3 / Dart, `flutter_secure_storage: ^10.0.0`, `crypto` (PBKDF2-HMAC-SHA256, transitive), `local_auth: ^3.0.0` (already in pubspec).

**Spec:** `docs/superpowers/specs/2026-08-29-pin-lock-design.md`

**Co-dependency:** Feature #1 (`PhoneOtpAuthProvider`) — "Forgot PIN" routes through the OTP flow.

## Global Constraints

- Dart SDK `^3.5.4`.
- PIN always 6 digits, exactly. Reject shorter/longer.
- Idle duration: `kIdleLockDuration = Duration(minutes: 2)`. Hard-coded constant.
- PBKDF2 with 100,000 iterations, SHA-256, 16-byte salt, 32-byte output.
- `flutter_secure_storage` keys: `pin_lock_v1`, `pin_lock_enabled_v1`.
- `shared_preferences` key: `lock_idle_enabled_v1`.
- Mandatory PIN setup on first login after this feature ships.
- Background lock (`AppLifecycleState.paused`) is always on; idle toggle does not affect it.
- After 5 wrong PIN attempts, force sign-out via `PhoneOtpAuthProvider.signOut()`.

```dart
import 'dart:convert';
import 'dart:math';
import 'package:crypto/crypto.dart';
import 'package:flutter_secure_storage/flutter_secure_storage.dart';

const _kKey = 'pin_lock_v1';
const _kIterations = 100000;
const _kPinLength = 6;
const _kSaltBytes = 16;

class PinService {
  PinService({FlutterSecureStorage? storage})
      : _storage = storage ?? const FlutterSecureStorage();
  final FlutterSecureStorage _storage;

  Future<bool> hasPin() async {
    final v = await _storage.read(key: _kKey);
    return v != null && v.isNotEmpty;
  }

  Future<void> setPin(String pin) {
    _validate(pin);
    return _write(pin);
  }

  Future<bool> verifyPin(String pin) async {
    _validate(pin);
    final stored = await _storage.read(key: _kKey);
    if (stored == null) return false;
    final data = jsonDecode(stored) as Map<String, dynamic>;
    final salt = base64Decode(data['salt'] as String);
    final expected = base64Decode(data['hash'] as String);
    final actual = await _pbkdf2(pin, salt, expected.length);
    return _constantTimeEquals(actual, expected);
  }

  Future<void> changePin({required String oldPin, required String newPin}) async {
    if (!await verifyPin(oldPin)) {
      throw StateError('Old PIN does not match');
    }
    _validate(newPin);
    await _write(newPin);
  }

  Future<void> clearPin() => _storage.delete(key: _kKey);

  void _validate(String pin) {
    if (pin.length != _kPinLength || !RegExp(r'^\d{6}$').hasMatch(pin)) {
      throw ArgumentError('PIN must be exactly 6 digits');
    }
  }

  Future<void> _write(String pin) async {
    final salt = _randomBytes(_kSaltBytes);
    final hash = await _pbkdf2(pin, salt, 32);
    final payload = jsonEncode({
      'salt': base64Encode(salt),
      'hash': base64Encode(hash),
      'iterations': _kIterations,
      'length': _kPinLength,
    });
    await _storage.write(key: _kKey, value: payload);
  }

  Future<List<int>> _pbkdf2(String pin, List<int> salt, int outLen) async {
    final hmac = Hmac(sha256, utf8.encode(pin));
    final blocks = (outLen / 32).ceil();
    final out = <int>[];
    for (var i = 1; i <= blocks; i++) {
      final block = await _pbkdf2Block(hmac, salt, i);
      out.addAll(block);
    }
    return out.sublist(0, outLen);
  }

  Future<List<int>> _pbkdf2Block(Hmac hmac, List<int> salt, int i) async {
    final seed = [...salt, ...[i >> 24 & 0xff, i >> 16 & 0xff, i >> 8 & 0xff, i & 0xff]];
    var u = hmac.convert(seed).bytes;
    final result = List<int>.from(u);
    for (var round = 1; round < _kIterations; round++) {
      u = hmac.convert(u).bytes;
      for (var j = 0; j < result.length; j++) {
        result[j] = result[j] ^ u[j];
      }
    }
    return result;
  }

  List<int> _randomBytes(int n) {
    final rng = Random.secure();
    return List<int>.generate(n, (_) => rng.nextInt(256));
  }

  bool _constantTimeEquals(List<int> a, List<int> b) {
    if (a.length != b.length) return false;
    var diff = 0;
    for (var i = 0; i < a.length; i++) {
      diff |= a[i] ^ b[i];
    }
    return diff == 0;
  }
}
```

```dart
import 'package:flutter/foundation.dart';
import '../services/pin_service.dart';

enum LockState { unlocked, locked, awaitingFirstSetup }

class LockStateProvider extends ChangeNotifier {
  LockStateProvider({required this.pinService, this.maxFailedAttempts = 5});
  final PinService pinService;
  final int maxFailedAttempts;

  LockState _state = LockState.unlocked;
  int _failedAttempts = 0;

  LockState get state => _state;
  int get failedAttempts => _failedAttempts;
  bool get shouldForceSignOut => _failedAttempts >= maxFailedAttempts;

  Future<void> initialize() async {
    if (!await pinService.hasPin()) {
      _state = LockState.awaitingFirstSetup;
    } else {
      _state = LockState.unlocked;
    }
    notifyListeners();
  }

  void lock() {
    _state = LockState.locked;
    _failedAttempts = 0;
    notifyListeners();
  }

  Future<bool> attemptUnlock(String pin) async {
    final ok = await pinService.verifyPin(pin);
    if (ok) {
      _state = LockState.unlocked;
      _failedAttempts = 0;
    } else {
      _failedAttempts += 1;
    }
    notifyListeners();
    return ok;
  }

  Future<void> reset() async {
    await pinService.clearPin();
    _state = LockState.awaitingFirstSetup;
    _failedAttempts = 0;
    notifyListeners();
  }
}
```

```dart
import 'dart:async';
import 'package:flutter/widgets.dart';
import '../providers/lock_state_provider.dart';

class IdleLockService with WidgetsBindingObserver {
  IdleLockService({required this.lockState, Duration? duration})
      : duration = duration ?? const Duration(minutes: 2) {
    assert(duration > Duration.zero);
  }
  final LockStateProvider lockState;
  final Duration duration;

  Timer? _timer;
  bool _attached = false;

  void attach() {
    if (_attached) return;
    WidgetsBinding.instance.addObserver(this);
    _attached = true;
    bump();
  }

  void detach() {
    if (!_attached) return;
    WidgetsBinding.instance.removeObserver(this);
    _attached = false;
    _timer?.cancel();
  }

  void bump() {
    _timer?.cancel();
    _timer = Timer(duration, () {
      if (lockState.state == LockState.unlocked) {
        lockState.lock();
      }
    });
  }

  void onUserInteraction() => bump();

  @override
  void didChangeAppLifecycleState(AppLifecycleState state) {
    if (state == AppLifecycleState.paused) {
      // Always lock on background
      if (lockState.state == LockState.unlocked) {
        lockState.lock();
      }
      _timer?.cancel();
    } else if (state == AppLifecycleState.resumed) {
      bump();
    }
  }
}
```

```dart
  final _digits = List.generate(6, (_) => TextEditingController());
  final _focus = List.generate(6, (_) => FocusNode());

  Future<void> _submit() async {
    final code = _digits.map((c) => c.text).join();
    if (code.length != 6) return;
    final lsp = context.read<LockStateProvider>();
    final ok = await lsp.attemptUnlock(code);
    final t = AppLocalizations.of(context)!;
    if (ok) {
      Navigator.of(context).pop();
      return;

    if (lsp.shouldForceSignOut) {
      ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text(t.pinTooMany)));
      await context.read<PhoneOtpAuthProvider>().signOut();
      Navigator.of(context).popUntil((r) => r.isFirst);
      return;
    }
    ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text(t.pinIncorrect)));
  }

    final auth = LocalAuthentication();
    final ok = await auth.authenticate(
      localizedReason: 'Unlock Cofiz',
      options: const AuthenticationOptions(biometricOnly: true),
    );
    if (ok && mounted) {
      Navigator.of(context).pop();
    }
```
